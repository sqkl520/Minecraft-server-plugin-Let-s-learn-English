package com.letslearnenglish.minecraftplugin.listener;

import org.bukkit.Bukkit;
import org.bukkit.ChatColor;
import org.bukkit.configuration.file.FileConfiguration;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.inventory.InventoryClickEvent;
import org.bukkit.event.inventory.InventoryCloseEvent;

import com.letslearnenglish.minecraftplugin.LetsLearnEnglish;
import com.letslearnenglish.minecraftplugin.core.SoundManager;
import com.letslearnenglish.minecraftplugin.core.word.Word;
import com.letslearnenglish.minecraftplugin.core.word.WordLearningManager;
import com.letslearnenglish.minecraftplugin.data.PlayerDataStore;
import com.letslearnenglish.minecraftplugin.gui.ConfirmationGUI;
import com.letslearnenglish.minecraftplugin.gui.EnglishSquareGUI;
import com.letslearnenglish.minecraftplugin.gui.MainMenu;
import com.letslearnenglish.minecraftplugin.gui.ProfileGUI;

import java.util.List;
import java.util.UUID;

public class GUIMenuListener implements Listener {

    private final LetsLearnEnglish plugin;

    public GUIMenuListener(LetsLearnEnglish plugin) {
        this.plugin = plugin;
    }

    public static boolean isPluginGUI(String title) {
        if (title == null) return false;
        return title.contains("Let's Learn English")
                || title.contains("一起学英语")
                || ConfirmationGUI.isConfirmGUI(title)
                || title.contains("Word Learning")
                || title.contains("单词学习")
                || title.contains("Dialogue")
                || title.contains("情景对话")
                || title.contains("Learning Progress")
                || title.contains("学习进度")
                || title.contains("English Square")
                || title.contains("英语广场")
                || title.contains("Personal Profile")
                || title.contains("'s Profile")
                || title.contains("的主页")
                || title.contains("个人主页")
                || title.contains("Privacy Settings")
                || title.contains("隐私设置");
    }

    @EventHandler
    public void onInventoryClose(InventoryCloseEvent event) {
        String title = event.getView().getTitle();

        if (!title.contains("Word Learning") && !title.contains("单词学习")) {
            return;
        }

        if (!(event.getPlayer() instanceof Player player)) {
            return;
        }

        WordLearningManager wlManager = plugin.getWordLearningManager();

        if (!wlManager.hasActiveQuiz(player.getUniqueId())) {
            return;
        }

        if (wlManager.isAnswering(player.getUniqueId())) {
            return;
        }

        UUID playerId = player.getUniqueId();
        Bukkit.getScheduler().runTaskLater(plugin, () -> {
            if (!wlManager.hasActiveQuiz(playerId)) return;
            if (wlManager.isAnswering(playerId)) return;
            String currentTitle = player.getOpenInventory().getTitle();
            if (isPluginGUI(currentTitle)) {
                return;
            }
            startAnswerForCurrentWord(player, wlManager);
        }, 2L);
    }

    @EventHandler
    public void onInventoryClick(InventoryClickEvent event) {
        String title = event.getView().getTitle();

        if (!isPluginGUI(title)) {
            return;
        }

        event.setCancelled(true);

        if (event.getCurrentItem() == null) {
            return;
        }

        if (!(event.getWhoClicked() instanceof Player player)) {
            return;
        }

        int slot = event.getSlot();

        if (title.contains("Let's Learn English") || title.contains("一起学英语")) {
            handleMainMenu(player, slot);
        } else if (ConfirmationGUI.isConfirmGUI(title)) {
            handleConfirmGUI(player, slot);
        } else if (title.contains("Word Learning") || title.contains("单词学习")) {
            handleWordLearningGUI(player, slot);
        } else if (title.contains("Dialogue") || title.contains("情景对话")) {
            handleDialogueGUI(player, slot);
        } else if (title.contains("Learning Progress") || title.contains("学习进度")) {
            handleProgressGUI(player, slot);
        } else if (title.contains("English Square") || title.contains("英语广场")) {
            handleEnglishSquareGUI(player, slot);
        } else if (title.contains("Personal Profile") || title.contains("'s Profile")
                || title.contains("的主页") || title.contains("个人主页")
                || title.contains("Privacy Settings")
                || title.contains("隐私设置")) {
            handleProfileGUI(player, slot);
        }
    }

    private void startAnswerForCurrentWord(Player player, WordLearningManager wlManager) {
        Word currentWord = wlManager.getCurrentWord(player.getUniqueId());
        if (currentWord == null) return;

        String lang = plugin.getPlayerLanguage(player);
        FileConfiguration messages = plugin.getConfigManager().getMessageConfig(lang);
        int timeout = wlManager.getAnswerTimeout();

        player.sendMessage(ChatColor.translateAlternateColorCodes('&',
                messages.getString("word-learning-gui.answer-prompt",
                        "&eSpell: &f{chinese} &7({seconds}s)")
                        .replace("{chinese}", currentWord.getChinese())
                        .replace("{seconds}", String.valueOf(timeout))));

        wlManager.startAnswerTimer(player);
    }

    private void handleMainMenu(Player player, int slot) {
        switch (slot) {
            case 11 -> {
                SoundManager.playSound(player, "menu-click");
                player.performCommand("le word");
            }
            case 13 -> {
                SoundManager.playSound(player, "menu-click");
                player.performCommand("le dialogue");
            }
            case 15 -> {
                SoundManager.playSound(player, "menu-click");
                player.performCommand("le progress");
            }
            case 17 -> {
                SoundManager.playSound(player, "menu-click");
                player.performCommand("le square");
            }
            case 18 -> {
                SoundManager.playSound(player, "menu-click");
                handleLanguageSwitch(player);
            }
            default -> {}
        }
    }

    private void handleLanguageSwitch(Player player) {
        String currentLang = plugin.getPlayerLanguage(player);
        String targetLang = "zh".equals(currentLang) ? "en" : "zh";
        ConfirmationGUI confirm = new ConfirmationGUI(plugin, targetLang);
        confirm.open(player);
    }

    private void handleConfirmGUI(Player player, int slot) {
        if (slot != 13) {
            return;
        }

        int stage = ConfirmationGUI.getStage(player);

        if (stage == 1) {
            String currentLang = plugin.getPlayerLanguage(player);
            String targetLang = "zh".equals(currentLang) ? "en" : "zh";
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                ConfirmationGUI confirm = new ConfirmationGUI(plugin, targetLang);
                confirm.openSecond(player);
            }, 2L);
        } else if (stage == 2) {
            String currentLang = plugin.getPlayerLanguage(player);
            String targetLang = "zh".equals(currentLang) ? "en" : "zh";
            ConfirmationGUI confirm = new ConfirmationGUI(plugin, targetLang);
            confirm.applySwitch(player);
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
        }
    }

    private void handleWordLearningGUI(Player player, int slot) {
        WordLearningManager wlManager = plugin.getWordLearningManager();

        if (slot == 16) {
            SoundManager.playSound(player, "menu-click");
            handleStopSession(player, wlManager);
        } else if (slot == 8) {
            SoundManager.playSound(player, "menu-click");
            wlManager.endQuiz(player.getUniqueId());
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
        } else if (slot == 22) {
            SoundManager.playSound(player, "menu-click");
            String lang = plugin.getPlayerLanguage(player);
            FileConfiguration messages = plugin.getConfigManager().getMessageConfig(lang);
            player.sendMessage(plugin.getMessageUtil().getPlayerMessage(player,
                    "word.session-ready"));
            player.sendMessage(ChatColor.translateAlternateColorCodes('&',
                    messages.getString("word-learning-gui.started-hint",
                            "&7Type the English word in chat after closing the GUI.")));
            startAnswerForCurrentWord(player, wlManager);
            player.closeInventory();
        }
    }

    private void handleStopSession(Player player, WordLearningManager wlManager) {
        String lang = plugin.getPlayerLanguage(player);
        FileConfiguration messages = plugin.getConfigManager().getMessageConfig(lang);

        wlManager.stopQuiz(player);
        player.closeInventory();

        player.sendMessage(ChatColor.translateAlternateColorCodes('&',
                messages.getString("word-learning-gui.session-ended",
                        "&aLearning session ended!")));
    }

    private void handleDialogueGUI(Player player, int slot) {
        if (slot == 22) {
            player.closeInventory();
        } else if (slot == 8) {
            SoundManager.playSound(player, "menu-click");
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
        }
    }

    private void handleProgressGUI(Player player, int slot) {
        if (slot == 8) {
            SoundManager.playSound(player, "menu-click");
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
        }
    }

    private void handleEnglishSquareGUI(Player player, int slot) {
        if (slot == 8) {
            SoundManager.playSound(player, "menu-click");
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
            return;
        }

        if (slot == 48) {
            SoundManager.playSound(player, "menu-click");
            String title = player.getOpenInventory().getTitle();
            int currentPage = extractPageFromTitle(title, 1);
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new EnglishSquareGUI(plugin).open(player, Math.max(1, currentPage - 1));
            }, 2L);
            return;
        }

        if (slot == 50) {
            SoundManager.playSound(player, "menu-click");
            String title = player.getOpenInventory().getTitle();
            int currentPage = extractPageFromTitle(title, 1);
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new EnglishSquareGUI(plugin).open(player, currentPage + 1);
            }, 2L);
            return;
        }

        if (slot >= 0 && slot < 45) {
            SoundManager.playSound(player, "menu-click");
            List<java.util.UUID> allPlayers = plugin.getPlayerDataStore().getActivePlayerIds();
            String invTitle = player.getOpenInventory().getTitle();
            int page = extractPageFromTitle(invTitle, 1);
            int startIdx = (page - 1) * 45;
            int playerIdx = startIdx + slot;
            if (playerIdx < allPlayers.size()) {
                java.util.UUID targetId = allPlayers.get(playerIdx);
                PlayerDataStore.ProfileSettings ps = plugin.getPlayerDataStore().getProfileSettings(targetId);
                player.closeInventory();
                if (ps.isShowInSquare()) {
                    Bukkit.getScheduler().runTaskLater(plugin, () -> {
                        new ProfileGUI(plugin).open(player, targetId);
                    }, 2L);
                } else {
                    Bukkit.getScheduler().runTaskLater(plugin, () -> {
                        new EnglishSquareGUI(plugin).openHidden(player, targetId);
                    }, 2L);
                }
            }
        }
    }

    private void handleProfileGUI(Player player, int slot) {
        if (slot == 49 || slot == 8) {
            SoundManager.playSound(player, "menu-click");
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new MainMenu(plugin).open(player);
            }, 2L);
            return;
        }

        if (slot == 51) {
            SoundManager.playSound(player, "menu-click");
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new ProfileGUI(plugin).openSettings(player);
            }, 2L);
            return;
        }

        if (slot == 22) {
            String ageLore = ChatColor.translateAlternateColorCodes('&',
                    plugin.getPlayerLanguage(player).equals("zh")
                            ? "&7不可以在这里设置年龄哦~"
                            : "&7You cannot set your age here~");
            player.sendMessage(ageLore);
            return;
        }

        String title = player.getOpenInventory().getTitle();
        if (title.contains("Privacy Settings") || title.contains("隐私设置")) {
            handlePrivacySettings(player, slot);
        }
    }

    private void handlePrivacySettings(Player player, int slot) {
        PlayerDataStore store = plugin.getPlayerDataStore();
        PlayerDataStore.ProfileSettings ps = store.getProfileSettings(player.getUniqueId());

        boolean changed = false;
        if (slot == 11) {
            ps.setShowAnswers(!ps.isShowAnswers());
            changed = true;
        } else if (slot == 13) {
            ps.setShowResults(!ps.isShowResults());
            changed = true;
        } else if (slot == 15) {
            ps.setShowInSquare(!ps.isShowInSquare());
            changed = true;
        }

        if (changed) {
            store.saveProfileSettings(player.getUniqueId());
            String lang = plugin.getPlayerLanguage(player);
            player.sendMessage(ChatColor.translateAlternateColorCodes('&',
                    lang.equals("zh") ? "&a设置已保存！" : "&aSettings saved!"));
            player.closeInventory();
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                new ProfileGUI(plugin).openSettings(player);
            }, 2L);
        }
    }

    private int extractPageFromTitle(String title, int defaultPage) {
        if (title == null) return defaultPage;
        java.util.regex.Pattern p = java.util.regex.Pattern.compile("Page (\\d+)");
        java.util.regex.Matcher m = p.matcher(title);
        if (m.find()) {
            try {
                return Integer.parseInt(m.group(1));
            } catch (NumberFormatException ignored) {}
        }
        return defaultPage;
    }
}
